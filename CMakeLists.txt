cmake_minimum_required(VERSION 3.16)
project(RoboMaster_Vision LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED NO_MODULE)

function(configure_target target)
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

add_library(predictor_filters STATIC
    src/predictor.cpp
    src/predictor_polar.cpp
    src/predictor_armor.cpp)
configure_target(predictor_filters)
target_include_directories(predictor_filters INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(predictor_filters PUBLIC Eigen3::Eigen)

add_library(armor_vision STATIC src/lightbar_detector.cpp src/solver.cpp)
configure_target(armor_vision)
target_include_directories(armor_vision INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(armor_vision PUBLIC ${OpenCV_LIBS} Eigen3::Eigen)

add_library(armor_video STATIC src/camera_tracking.cpp)
configure_target(armor_video)
target_compile_definitions(armor_video PRIVATE CAMERA_TRACKING_LIBRARY)
target_link_libraries(armor_video PUBLIC armor_vision predictor_filters)

add_executable(Armor src/Armor.cpp)
target_link_libraries(Armor PRIVATE armor_video)
target_compile_definitions(Armor PRIVATE ARMOR_PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(camera_tracking src/camera_tracking.cpp)
target_link_libraries(camera_tracking PRIVATE armor_vision predictor_filters)

foreach(target pose_base)
    add_executable(${target} src/${target}.cpp)
    target_link_libraries(${target} PRIVATE Eigen3::Eigen)
    target_compile_definitions(${target} PRIVATE CAMERA_DEFAULT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
    if(MINGW)
        target_link_options(${target} PRIVATE -static)
    endif()
endforeach()

foreach(target Armor pose_base camera_tracking)
    configure_target(${target})
    # A generator expression keeps multi-config builds in the same output directory.
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/$<0:>")
endforeach()
