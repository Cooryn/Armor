cmake_minimum_required(VERSION 3.16)
project(RoboMaster_Vision LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(Threads REQUIRED)

function(configure_target target)
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

add_library(serial STATIC src/serial.cpp)
configure_target(serial)
target_include_directories(serial INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(serial PUBLIC Threads::Threads)

add_library(predictor_filters STATIC
    src/predictor.cpp
    src/predictor_polar.cpp
    src/predictor_armor.cpp)
configure_target(predictor_filters)
target_include_directories(predictor_filters INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(predictor_filters PUBLIC Eigen3::Eigen)

add_library(armor_vision STATIC src/lightbar_detector.cpp src/solver.cpp)
configure_target(armor_vision)
target_include_directories(armor_vision INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(armor_vision PUBLIC ${OpenCV_LIBS} Eigen3::Eigen)

add_library(armor_pipeline STATIC src/camera.cpp src/pose_base.cpp src/gimbal.cpp src/output.cpp)
configure_target(armor_pipeline)
target_link_libraries(armor_pipeline PUBLIC armor_vision predictor_filters serial)

add_executable(Armor src/Armor.cpp)
configure_target(Armor)
target_link_libraries(Armor PRIVATE armor_pipeline)
target_compile_definitions(Armor PRIVATE ARMOR_PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(Armor PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/$<0:>")

add_executable(Armor_detect src/Armor_detect.cpp src/camera.cpp)
configure_target(Armor_detect)
target_link_libraries(Armor_detect PRIVATE armor_vision serial)
target_compile_definitions(Armor_detect PRIVATE ARMOR_PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(Armor_detect PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/$<0:>")
